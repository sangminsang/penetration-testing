# 1) ZAP 설치 경로로 이동
cd "D:\ZAP\Zed Attack Proxy"

# 2) JVM 메모리 설정 (1GB에서 4GB로 증설)
# ZAP은 스캔 데이터를 메모리에 쌓기 때문에 4GB 정도가 적당합니다.
$env:_JAVA_OPTIONS="-Xmx4096m"

# 3) ZAP 최적화 실행 명령어
& ".\zap.bat" `
    -daemon `
    -port 8080 `
    -host 0.0.0.0 `
    -config api.key=change-me-9203935709 `
    -config api.addrs.addr.name=.* `
    -config api.addrs.addr.regex=true `
    `
    # --- [추가된 최적화 설정] --- `
    `
    # [스파이더 무한 루프 방지] `
    -config spider.maxDepth=10 `
    -config spider.maxChildren=100 `
    `
    # [불필요한 리소스 제외 - 메모리 절약] `
    # node_modules와 무한 반복되는 assets 경로를 스캔에서 제외합니다. `
    -config globalexcludeurl.urlList.url.regex=true `
    -config globalexcludeurl.urlList.url.url="^.*node_modules.*$" `
    -config globalexcludeurl.urlList.url.url="^.*(?:assets/public/assets).*$" `
    `
    # [연산량 감소] `
    -config spider.parseComments=false `
    -config spider.processForm=false `
    `
    # [대상 서버 과부하 방지] `
    # 너무 빠른 요청으로 Juice Shop이 뻗는 것을 방지하기 위해 스레드 수를 제한합니다. `
    -config spider.threadCount=5
